import Foundation
func t(_ f: () -> Void) -> Double { let s = DispatchTime.now().uptimeNanoseconds; f(); return Double(DispatchTime.now().uptimeNanoseconds - s) / 1e6 }

// 1. Catastrophic pattern, no guard: how long does n take? (grows exponentially)
let evil = try! NSRegularExpression(pattern: "^(a+)+$")
for n in [18, 22, 24, 26] {
    let s = String(repeating: "a", count: n) + "b"
    let ms = t { _ = evil.firstMatch(in: s, range: NSRange(s.startIndex..., in: s)) }
    print("unguarded n=\(n): \(String(format: "%.1f", ms)) ms")
}
// 2. Does .reportProgress let us stop it, and how often does the block fire?
func guarded(_ s: String, maxCalls: Int) -> (calls: Int, stopped: Bool, ms: Double) {
    var calls = 0; var stopped = false
    let ms = t {
        evil.enumerateMatches(in: s, options: [.reportProgress], range: NSRange(s.startIndex..., in: s)) { _, flags, stop in
            if flags.contains(.progress) { calls += 1; if calls >= maxCalls { stopped = true; stop.pointee = true } }
        }
    }
    return (calls, stopped, ms)
}
for n in [20, 26, 40, 200] {
    let s = String(repeating: "a", count: n) + "b"
    let r = guarded(s, maxCalls: 50)
    print("guarded n=\(n) maxCalls=50: calls=\(r.calls) stopped=\(r.stopped) \(String(format: "%.2f", r.ms)) ms")
}
// 3. Is the progress count a function of work (deterministic), not of time? Same input twice.
let s = String(repeating: "a", count: 22) + "b"
print("repeat:", guarded(s, maxCalls: 1_000_000).calls, guarded(s, maxCalls: 1_000_000).calls)
// 4. A benign match: how many progress calls on a 3.7 MB-ish input?
let big = String(repeating: "let x = 1\n", count: 400_000)
let benign = try! NSRegularExpression(pattern: "zzz")
var c = 0
let ms = t { benign.enumerateMatches(in: big, options: [.reportProgress], range: NSRange(big.startIndex..., in: big)) { _, f, _ in if f.contains(.progress) { c += 1 } } }
print("benign 4 MB scan: progress calls=\(c) \(String(format: "%.1f", ms)) ms")

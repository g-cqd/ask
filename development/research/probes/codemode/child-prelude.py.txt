import asyncio, json, sys
pending = {}; counter = 0
def send(m): sys.stdout.write(json.dumps(m) + "\n"); sys.stdout.flush()
class Tools:
    def __getattr__(self, name):
        def call(**args):
            global counter
            counter += 1
            future = asyncio.get_running_loop().create_future()
            pending[counter] = future
            send({"call": counter, "tool": name, "args": args})
            return future
        return call
async def main():
    loop = asyncio.get_running_loop()
    reader = asyncio.StreamReader()
    await loop.connect_read_pipe(lambda: asyncio.StreamReaderProtocol(reader), sys.stdin)
    send({"ready": True})
    running = None
    while True:
        line = await reader.readline()
        if not line: break
        m = json.loads(line)
        if "script" in m:
            body = "\n".join("    " + l for l in m["script"].split("\n"))
            scope = {"tools": Tools(), "text": lambda t: send({"text": str(t)}), "asyncio": asyncio, "json": json}
            exec("async def __script():\n" + body, scope)
            async def run():
                try: send({"done": await scope["__script"]()})
                except Exception as e: send({"failed": repr(e)})
                sys.exit(0)
            running = asyncio.ensure_future(run())
        else:
            f = pending.pop(m["id"])
            f.set_exception(RuntimeError(m["error"])) if m.get("error") else f.set_result(m["result"])
asyncio.run(main())

import json, os, subprocess, sys, time, statistics

def facts(python):
    out = subprocess.run([python, "-I", "-S", "-c",
        "import sys, os; print(os.path.realpath(sys.executable)); print(os.path.realpath(sys.base_prefix)); print(os.path.realpath(sys.base_exec_prefix)); print(sys.version_info[0], sys.version_info[1])"],
        capture_output=True, text=True).stdout.split("\n")
    return {"executable": out[0], "prefix": out[1], "exec_prefix": out[2], "version": out[3]}

def ancestors(path):
    parts = []
    while path not in ("/", ""):
        path = os.path.dirname(path)
        parts.append(path)
    return parts

def profile(f, extra=""):
    roots = sorted({f["prefix"], f["exec_prefix"]})
    up = sorted({a for r in roots + [f["executable"]] for a in ancestors(r)})
    lines = ["(version 1)", "(deny default)",
             "(allow process-exec " + " ".join(f'(subpath "{r}")' for r in roots) + f' (literal "{f["executable"]}"))',
             "(allow file-read* " + " ".join(f'(subpath "{r}")' for r in roots) + ")",
             "(allow file-read-metadata " + " ".join(f'(literal "{a}")' for a in up) + ")",
             extra]
    return "\n".join(l for l in lines if l)

def run(python, prof, code, n=1, stdin=None):
    f = facts(python)
    ts = []
    for _ in range(n):
        t = time.monotonic()
        r = subprocess.run(["/usr/bin/sandbox-exec", "-p", prof, f["executable"], "-I", "-S", "-c", code], capture_output=True, text=True, env={}, input=stdin)
        ts.append((time.monotonic() - t) * 1000)
    return r.returncode, (r.stdout.strip() or r.stderr.strip().split("\n")[-1])[:160], statistics.median(ts)

if __name__ == "__main__":
    python = sys.argv[1]
    f = facts(python); print(json.dumps(f))
    base = profile(f)
    print("--- bare deny-default with the prefix only:")
    print(run(python, base, "print('started')"))

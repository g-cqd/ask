func ms(_ d: Duration) -> String { String(format: "%.2f ms", Double(d.components.attoseconds) / 1e15 + Double(d.components.seconds) * 1000) }
func rss() -> Int {
    var info = mach_task_basic_info(); var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size / MemoryLayout<natural_t>.size)
    _ = withUnsafeMutablePointer(to: &info) { $0.withMemoryRebound(to: integer_t.self, capacity: Int(count)) { task_info(mach_task_self_, task_flavor_t(MACH_TASK_BASIC_INFO), $0, &count) } }
    return Int(info.resident_size) / 1024
}

// The bridge: tools.call(name, args) returns a Promise that Swift settles later, from another queue.
let queue = DispatchQueue(label: "tools", attributes: .concurrent)
nonisolated(unsafe) var calls = 0
let call: @convention(block) (String, JSValue) -> JSValue = { name, args in
    calls += 1
    let text = args.toString() ?? ""
    return JSValue(newPromiseIn: context) { resolve, _ in
        queue.asyncAfter(deadline: .now() + .milliseconds(name == "slow" ? 30 : 0)) {
            resolve!.call(withArguments: ["\(name):\(text)"])
        }
    }
}
context.setObject(call, forKeyedSubscript: "__call" as NSString)
nonisolated(unsafe) var output: [String] = []
let text: @convention(block) (JSValue) -> Void = { output.append($0.toString() ?? "") }
context.setObject(text, forKeyedSubscript: "text" as NSString)
let done = DispatchSemaphore(value: 0)
nonisolated(unsafe) var outcome = ""
let finish: @convention(block) (String, JSValue) -> Void = { kind, value in outcome = "\(kind): \(value.toString() ?? "")"; done.signal() }
context.setObject(finish, forKeyedSubscript: "__finish" as NSString)
context.exceptionHandler = { _, error in outcome = "exception: \(error?.toString() ?? "?")"; done.signal() }

func run(_ body: String, wait: Double = 5) -> (String, Duration) {
    outcome = "(nothing)"; output = []
    let took = clock.measure {
        context.evaluateScript("""
            globalThis.tools = new Proxy({}, { get: (_, name) => (args) => __call(String(name), JSON.stringify(args ?? null)) });
            (async function () { \(body)
            })().then((v) => __finish("returned", v === undefined ? "" : JSON.stringify(v)), (e) => __finish("threw", String(e && e.stack || e)));
            """)
        _ = done.wait(timeout: .now() + wait)
    }
    return (outcome, took)
}

if mode == "spin" {
    // Can a script that never yields be stopped? Public API has no way; try the private time limit.
    typealias SetLimit = @convention(c) (JSContextGroupRef, Double, (@convention(c) (JSContextRef?, UnsafeMutableRawPointer?) -> Bool)?, UnsafeMutableRawPointer?) -> Void
    let symbol = dlsym(UnsafeMutableRawPointer(bitPattern: -2), "JSContextGroupSetExecutionTimeLimit")
    print("JSContextGroupSetExecutionTimeLimit exported: \(symbol != nil)")
    if let symbol {
        let set = unsafeBitCast(symbol, to: SetLimit.self)
        set(JSContextGetGroup(context.jsGlobalContextRef), 0.2, nil, nil)
        let took = clock.measure { context.evaluateScript("while (true) {}") }
        print("while(true){} with a 0.2 s limit: returned after \(ms(took)); exception: \(context.exception?.toString() ?? "none")")
        context.exception = nil
        let again = clock.measure { _ = context.evaluateScript("1 + 1") }
        print("the context afterwards: 1+1 = \(context.evaluateScript("1 + 1")!.toInt32()) (\(ms(again)))")
    }
}

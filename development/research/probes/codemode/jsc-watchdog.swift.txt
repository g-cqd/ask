import Foundation
import JavaScriptCore

func rssMiB() -> Int {
    var info = mach_task_basic_info(); var count = mach_msg_type_number_t(MemoryLayout<mach_task_basic_info>.size / MemoryLayout<natural_t>.size)
    _ = withUnsafeMutablePointer(to: &info) { $0.withMemoryRebound(to: integer_t.self, capacity: Int(count)) { task_info(mach_task_self_, task_flavor_t(MACH_TASK_BASIC_INFO), $0, &count) } }
    return Int(info.resident_size) / 1_048_576
}
typealias Callback = @convention(c) (JSContextRef?, UnsafeMutableRawPointer?) -> Bool
typealias SetLimit = @convention(c) (JSContextGroupRef, Double, Callback?, UnsafeMutableRawPointer?) -> Void
let set = unsafeBitCast(dlsym(UnsafeMutableRawPointer(bitPattern: -2), "JSContextGroupSetExecutionTimeLimit")!, to: SetLimit.self)
nonisolated(unsafe) var baseline = 0
nonisolated(unsafe) var deadline = ContinuousClock.now
nonisolated(unsafe) var why = ""
nonisolated(unsafe) var polls = 0
nonisolated(unsafe) var again: Callback? = nil
let watch: Callback = { context, _ in
    polls += 1
    if rssMiB() - baseline > 256 { why = "memory: \(rssMiB() - baseline) MiB over the start"; return true }
    if ContinuousClock.now > deadline { why = "deadline"; return true }
    // Not asked again by itself: the limit is armed once.
    set(JSContextGetGroup(JSContextGetGlobalContext(context!)), 0.02, again, nil)
    return false
}
again = watch
let clock = ContinuousClock()
let only = CommandLine.arguments.dropFirst().first
Thread.detachNewThread { while true { usleep(20_000); if rssMiB() > 1500 { print("GUARD: rss \(rssMiB()) MiB, polls \(polls), why '\(why)'"); exit(9) } } }
for (label, script, seconds) in [("allocating without end", "const big = []; for (;;) big.push(new Array(1 << 17).fill(1));", 10.0), ("spinning", "while (true) {}", 0.5),
                                 ("catching the stop", "for (;;) { try { while (true) {} } catch (e) {} }", 0.5), ("an honest script", "let s = 0; for (let i = 0; i < 3e7; i++) s += i; s", 5.0)] {
    if let only, !label.hasPrefix(only) { continue }
    let context = JSContext()!
    baseline = rssMiB(); deadline = .now + .milliseconds(Int(seconds * 1000)); why = ""; polls = 0
    set(JSContextGetGroup(context.jsGlobalContextRef), 0.02, watch, nil)
    var value: JSValue?
    let took = clock.measure { value = context.evaluateScript(script) }
    let ms = Double(took.components.attoseconds) / 1e15 + Double(took.components.seconds) * 1000
    print("\(label): stopped=\(why.isEmpty ? "no" : why) after \(String(format: "%.0f", ms)) ms, \(polls) polls, rss +\(rssMiB() - baseline) MiB, value \(value?.isUndefined == false ? value!.toString()! : "none")")
}
